function renderAgingResults(data) {
    window.lastAgingResult = data;
    // Reveal Cards
    document.getElementById('aging-eol-card').style.display = 'block';
    document.getElementById('aging-autopsy-card').style.display = 'block';
    document.getElementById('aging-highlights-card').style.display = 'block';
    document.getElementById('aging-chart-card').style.display = 'block';
    document.getElementById('aging-mechanism-card').style.display = 'block';
    document.getElementById('aging-table-card').style.display = 'block';

    const series = data.aging_profile;
    const final = series[series.length - 1] || {};
    const eol_reached = data.end_of_life_month !== null;
    const eol_month = data.end_of_life_month || '> ' + document.getElementById('aging-design-life').value;
    const dominant_mech = data.dominant_mechanism || 'N/A';

    // Highlights
    document.getElementById('aging-hl-npf').textContent = final.npf ? final.npf.toFixed(3) : '-';
    document.getElementById('aging-hl-pfeed').textContent = final.p_feed_bar ? final.p_feed_bar.toFixed(1) : '-';
    document.getElementById('aging-hl-cips').textContent = data.cip_events ? data.cip_events.length : 0;
    document.getElementById('aging-eol-month').textContent = eol_month;
    document.getElementById('aging-hl-eol').textContent = eol_month;
    document.getElementById('aging-dominant-mech').textContent = dominant_mech.replace('_', ' ').toUpperCase();

    // Table
    const tbody = document.getElementById('aging-monthly-tbody');
    tbody.innerHTML = '';
    series.forEach(pt => {
        const tr = document.createElement('tr');
        tr.innerHTML = `
            <td>${pt.month}</td>
            <td style="color: var(--accent); font-weight: 600;">${pt.p_feed_bar.toFixed(2)}</td>
            <td>${pt.npf.toFixed(2)}</td>
            <td>${pt.nsr.toFixed(2)}</td>
            <td>${pt.delta_p_ratio.toFixed(2)}</td>
            <td>${pt.flux_lmh.toFixed(1)}</td>
            <td>${pt.recovery_pct.toFixed(1)}</td>
            <td>${pt.cip_event ? 'Γ£à' : ''}</td>
        `;
        tbody.appendChild(tr);
    });

    // Autopsy Table
    const autoTbody = document.getElementById('aging-autopsy-tbody');
    autoTbody.innerHTML = '';
    const finalElems = data.element_autopsy;
    
    // Check if we have element-level data
    if (finalElems && typeof finalElems === 'object') {
        Object.entries(finalElems).forEach(([key, el]) => {
            const tr = document.createElement('tr');
            const name = key.replace('s', 'Stage ').replace('_e', ', Element ');
            tr.innerHTML = `
                <td>${name}</td>
                <td>${el.fri_cake.toFixed(3)}</td>
                <td>${el.fri_bio.toFixed(3)}</td>
                <td>${el.fri_nom.toFixed(3)}</td>
                <td>${el.fri_scale.toFixed(3)}</td>
                <td style="font-weight: bold;">${el.fri_total.toFixed(3)}</td>
                <td>${(el.a_eff * 100).toFixed(1)}%</td>
            `;
